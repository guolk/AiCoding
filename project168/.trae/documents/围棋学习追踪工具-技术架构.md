## 2. 技术选型

- **前端框架**: React@18 + TypeScript
- **构建工具**: Vite@5
- **样式方案**: TailwindCSS@3
- **路由管理**: React Router@6
- **状态管理**: Zustand@4
- **图表库**: Recharts
- **图标库**: Lucide React
- **日期处理**: date-fns
- **数据持久化**: LocalStorage（配置数据） + IndexedDB（大体积SGF文件）
- **本地存储封装: nanoid 唯一ID生成

## 3. 路由定义

| 路由 | 页面 | 页面名称 | 页面功能 |
|------|--------|----------|
| /dashboard | 仪表板主页 | 学习概览、今日任务、快捷入口、最近活动 |
| /games | 棋谱管理页 | 棋谱列表、导入、分类标签、变化图管理 |
| /games/:id | 棋局回放页 | 棋盘展示、手顺控制、标注、形势说明、变化图切换 |
| /learning | 学习进度页 | 死活题、定式掌握、每日任务、学习统计 |
| /records | 对局记录页 | 对局存档列表、复盘笔记、段位历史 |

## 4. API 定义

本应用为纯前端单页应用，所有数据通过封装接口定义：

```typescript
// 棋谱相关类型定义：

// 棋子颜色
type StoneColor = 'black' | 'white';

// 棋盘坐标
interface Point = {
  x: number;
  y: number;
};

// 棋谱节点
interface MoveNode {
  id: string;
  moveNumber: number;
  color: StoneColor;
  point: Point | null; // null表示pass
  comment?: string;
  marks?: MoveMark[];
  children: MoveNode[];
  parentId: string | null;
  isMain?: boolean;
}

// 手顺标注类型
type MarkType = 'key' | 'doubt' | 'good' | 'custom';

interface MoveMark {
  id: string;
  type: MarkType;
  text?: string;
}

// 棋谱分类
type GameCategory = 'joseki' | 'problem' | 'famous' | 'self' | 'teaching' | 'custom';

// 棋谱
interface GameRecord {
  id: string;
  title: string;
  blackPlayer: string;
  whitePlayer: string;
  result?: string;
  date: string;
  category: GameCategory;
  tags: string[];
  description?: string;
  sgfContent: string;
  rootNode: MoveNode;
  currentNode: MoveNode;
  createdAt: number;
  updatedAt: number;
}

// 死活题难度
type ProblemDifficulty = 'easy' | 'medium' | 'hard' | 'expert';

// 死活题
interface LifeDeathProblem {
  id: string;
  title: string;
  difficulty: ProblemDifficulty;
  initialBoard: string; // SGF格式的初始局面
  correctAnswer: string; // SGF格式的正确答案
  practiceRecords: PracticeRecord[];
}

// 练习记录
interface PracticeRecord {
  id: string;
  date: number;
  isCorrect: boolean;
  timeSpent: number;
  notes?: string;
}

// 定式掌握程度
type JosekiMastery = 'familiar' | 'memorized' | 'understood';

// 定式
interface Joseki {
  id: string;
  name: string;
  gameId: string; // 关联的棋谱ID
  mastery: JosekiMastery;
  lastPracticedAt?: number;
}

// 每日任务类型
type DailyTaskType = 'problem' | 'joseki' | 'game' | 'custom';

// 每日任务状态
type DailyTask {
  id: string;
  date: string; // YYYY-MM-DD
  type: DailyTaskType;
  title: string;
  isCompleted: boolean;
  completedAt?: number;
  targetCount?: number;
  currentCount?: number;
}

// 对局信息
type GameResult = 'win' | 'loss' | 'draw' | 'unknown';

// 对手等级
type OpponentRank {
  level: number;
  rank: string;
}

// 对局记录
interface MatchRecord {
  id: string;
  opponentName: string;
  opponentRank: string;
  result: GameResult;
  myColor: StoneColor;
  handicap: number; // 让子数
  playedAt: number;
  reviewNotes?: string;
  keyMoments: KeyMoment[];
  gameRecordId?: string;
}

// 关键时刻
interface KeyMoment {
  id: string;
  moveNumber: number;
  description: string;
  reflection?: string;
}

// 段位记录
interface RankRecord {
  id: string;
  rank: string;
  date: number;
  event?: string;
  notes?: string;
}

// 学习统计
interface LearningStats {
  totalStudyTime: number;
  totalMatches: number;
  winRate: number;
  currentStreak: number;
  rank: RankRecord | null;
}
```
